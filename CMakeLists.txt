cmake_minimum_required(VERSION 3.20)
project(MatMul VERSION 1.0.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(MATMUL_OPT_LEVEL "3" CACHE STRING "Release optimization level (0, 2, 3)")
set_property(CACHE MATMUL_OPT_LEVEL PROPERTY STRINGS 0 2 3)
if(NOT MATMUL_OPT_LEVEL MATCHES "^(0|2|3)$")
  message(FATAL_ERROR "MATMUL_OPT_LEVEL must be 0, 2, or 3")
endif()
option(MATMUL_FAST_MATH "Explicitly enable relaxed floating-point math" OFF)
option(MATMUL_VECTORIZATION_REPORT "Enable compiler vectorization diagnostics" OFF)
option(MATMUL_DISABLE_VECTORIZATION "Disable automatic loop and SLP vectorization for ablation" OFF)
option(MATMUL_ASAN_UBSAN "Enable AddressSanitizer and UndefinedBehaviorSanitizer" OFF)
option(MATMUL_TSAN "Enable ThreadSanitizer (separate build)" OFF)
if(MATMUL_ASAN_UBSAN AND MATMUL_TSAN)
  message(FATAL_ERROR "ThreadSanitizer and AddressSanitizer require separate builds")
endif()

find_package(Threads REQUIRED)
add_library(matmul_lib
  src/matrix.cpp src/verification.cpp src/algorithms.cpp src/naive.cpp
  src/reordered.cpp src/blocked.cpp src/parallel.cpp src/accelerate.cpp src/benchmark.cpp src/profiling.cpp)
target_include_directories(matmul_lib PUBLIC include PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated/$<CONFIG>")
target_link_libraries(matmul_lib PUBLIC Threads::Threads)
target_compile_options(matmul_lib PRIVATE -Wall -Wextra -Wpedantic)
target_compile_options(matmul_lib PUBLIC
  "$<$<OR:$<CONFIG:Release>,$<CONFIG:RelWithDebInfo>>:-O${MATMUL_OPT_LEVEL}>")
set(MATMUL_VECTOR_FLAGS "")
if(MATMUL_DISABLE_VECTORIZATION)
  # These must follow -O3: Clang can re-enable vectorization when -O3 follows
  # disabling flags supplied through the earlier CMAKE_CXX_FLAGS variable.
  if(CMAKE_CXX_COMPILER_ID MATCHES "Clang")
    set(MATMUL_VECTOR_FLAGS "-fno-vectorize -fno-slp-vectorize")
    target_compile_options(matmul_lib PRIVATE -fno-vectorize -fno-slp-vectorize)
  elseif(CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
    set(MATMUL_VECTOR_FLAGS "-fno-tree-vectorize")
    target_compile_options(matmul_lib PRIVATE -fno-tree-vectorize)
  else()
    message(FATAL_ERROR "Vectorization ablation requires Clang or GCC")
  endif()
endif()
if(MATMUL_FAST_MATH)
  # Relax only arithmetic kernels. Validation, the FP64 oracle, statistics and
  # tests retain IEEE checks (fast-math otherwise erases isfinite checks).
  set_source_files_properties(src/naive.cpp src/reordered.cpp src/blocked.cpp
    src/parallel.cpp PROPERTIES COMPILE_OPTIONS "-ffast-math")
endif()
if(MATMUL_VECTORIZATION_REPORT)
  if(CMAKE_CXX_COMPILER_ID MATCHES "Clang")
    target_compile_options(matmul_lib PRIVATE -Rpass=loop-vectorize -Rpass-missed=loop-vectorize -Rpass-analysis=loop-vectorize)
  elseif(CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
    target_compile_options(matmul_lib PRIVATE -fopt-info-vec-optimized -fopt-info-vec-missed)
  endif()
endif()
if(MATMUL_ASAN_UBSAN)
  target_compile_options(matmul_lib PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(matmul_lib PUBLIC -fsanitize=address,undefined)
endif()
if(MATMUL_TSAN)
  target_compile_options(matmul_lib PUBLIC -fsanitize=thread -fno-omit-frame-pointer)
  target_link_options(matmul_lib PUBLIC -fsanitize=thread)
endif()
if(APPLE)
  find_library(ACCELERATE_FRAMEWORK Accelerate REQUIRED)
  target_link_libraries(matmul_lib PUBLIC "${ACCELERATE_FRAMEWORK}")
  target_compile_definitions(matmul_lib PRIVATE MATMUL_HAS_ACCELERATE=1)
endif()

set(MATMUL_COMPILER "${CMAKE_CXX_COMPILER_ID} ${CMAKE_CXX_COMPILER_VERSION}")
set(MATMUL_CONFIG_FLAGS "")
foreach(CONFIG Debug Release RelWithDebInfo MinSizeRel)
  string(TOUPPER "${CONFIG}" CONFIG_UPPER)
  string(APPEND MATMUL_CONFIG_FLAGS "$<$<CONFIG:${CONFIG}>:${CMAKE_CXX_FLAGS_${CONFIG_UPPER}}>")
endforeach()
set(MATMUL_FLAGS "${CMAKE_CXX_FLAGS} ${MATMUL_CONFIG_FLAGS} $<$<OR:$<CONFIG:Release>,$<CONFIG:RelWithDebInfo>>:-O${MATMUL_OPT_LEVEL}> ${MATMUL_VECTOR_FLAGS}; kernel_fast_math=${MATMUL_FAST_MATH}; asan_ubsan=${MATMUL_ASAN_UBSAN}; tsan=${MATMUL_TSAN}")
set(MATMUL_BUILD_TYPE "$<CONFIG>")
foreach(META MATMUL_COMPILER MATMUL_FLAGS CMAKE_SYSTEM_PROCESSOR MATMUL_BUILD_TYPE)
  string(REPLACE "\\" "\\\\" ${META} "${${META}}")
  string(REPLACE "\"" "\\\"" ${META} "${${META}}")
endforeach()
configure_file(cmake/build_info.hpp.in generated/build_info.config.hpp @ONLY)
file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/generated/$<CONFIG>/build_info.hpp"
  INPUT "${CMAKE_CURRENT_BINARY_DIR}/generated/build_info.config.hpp")
add_executable(matmul src/main.cpp)
target_link_libraries(matmul PRIVATE matmul_lib)
include(CTest)
if(BUILD_TESTING)
  file(GLOB TEST_SOURCES CONFIGURE_DEPENDS tests/*.cpp)
  if(TEST_SOURCES)
    add_executable(matmul_tests ${TEST_SOURCES})
    target_link_libraries(matmul_tests PRIVATE matmul_lib)
    add_test(NAME matmul_unit COMMAND matmul_tests)
  endif()
  add_test(NAME matmul_verify COMMAND matmul verify)
  add_test(NAME matmul_cli_smoke COMMAND matmul benchmark --implementation blocked --rows 7 --inner 9 --cols 5 --block-size 4 --repetitions 3 --warmups 1 --json)
  add_test(NAME matmul_cli_reject_zero_repetitions COMMAND matmul benchmark --repetitions 0)
  set_tests_properties(matmul_cli_reject_zero_repetitions PROPERTIES WILL_FAIL TRUE)
  find_package(Python3 3.10 COMPONENTS Interpreter QUIET)
  if(Python3_Interpreter_FOUND)
    add_test(NAME matmul_cli_validation
      COMMAND "${CMAKE_COMMAND}" -E env "MATMUL_BINARY=$<TARGET_FILE:matmul>"
      "${Python3_EXECUTABLE}" -m unittest discover -s tests -p test_cli.py)
    set_tests_properties(matmul_cli_validation PROPERTIES WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}")
  endif()
endif()
install(TARGETS matmul RUNTIME DESTINATION bin)
